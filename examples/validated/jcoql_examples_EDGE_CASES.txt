==============================================
JCOQL+ EXAMPLES - EDGE CASES & COMPLEX QUERIES
==============================================

ESEMPIO 1: Condizioni Multiple con Parentesi Complesse
----------------------------------------------
Descrizione: Query con logica booleana complessa
Query:
GET COLLECTION products;
FILTER CASE 
    WHERE ( ( .category = 'electronics' OR .category = 'tech' ) AND .price > 100 )
    OR ( .premium = true AND .stock > 0 )
KEEP OTHERS;

ESEMPIO 2: Range con Valori Negativi
----------------------------------------------
Descrizione: Temperature sotto zero
Query:
GET COLLECTION weather;
FILTER CASE WHERE .temperature IN_RANGE [ -10 , 5 ]
KEEP OTHERS;

ESEMPIO 3: Multiple IN_RANGE con OR
----------------------------------------------
Descrizione: Più intervalli disgiunti
Query:
GET COLLECTION products;
FILTER CASE 
    WHERE .price IN_RANGE [ 10 , 50 ] OR .price IN_RANGE [ 100 , 200 ]
KEEP OTHERS;

ESEMPIO 4: Field Names con Caratteri Speciali
----------------------------------------------
Descrizione: Campi con underscore
Query:
GET COLLECTION users;
FILTER CASE WHERE .first_name = 'Mario' AND .last_name = 'Rossi'
KEEP OTHERS;

ESEMPIO 5: Stringhe Vuote
----------------------------------------------
Descrizione: Controllo campo vuoto
Query:
GET COLLECTION docs;
FILTER CASE WHERE NOT .title = ''
KEEP OTHERS;

ESEMPIO 6: Combinazione IS NULL e Condizioni
----------------------------------------------
Descrizione: NULL check con altre condizioni
Query:
GET COLLECTION orders;
FILTER CASE 
    WHERE .status = 'pending' AND .notes IS NULL
KEEP OTHERS;

ESEMPIO 7: WITHOUT su Campo Opzionale
----------------------------------------------
Descrizione: Verifica assenza campo opzionale
Query:
GET COLLECTION profiles;
FILTER CASE WHERE WITHOUT .avatar
KEEP OTHERS;

ESEMPIO 8: WITH ARRAY
----------------------------------------------
Descrizione: Verifica presenza array
Query:
GET COLLECTION users;
FILTER CASE WHERE WITH ARRAY .tags
KEEP OTHERS;

ESEMPIO 9: Negazione Complessa
----------------------------------------------
Descrizione: NOT con condizioni multiple
Query:
GET COLLECTION products;
FILTER CASE 
    WHERE NOT ( .discontinued = true OR .stock = 0 )
KEEP OTHERS;

ESEMPIO 10: Valori Decimali Precisi
----------------------------------------------
Descrizione: Confronto con decimali
Query:
GET COLLECTION measurements;
FILTER CASE WHERE .value > 3.14159
KEEP OTHERS;

ESEMPIO 11: Range Estremi Esclusi
----------------------------------------------
Descrizione: Parentesi tonde per escludere estremi
Query:
GET COLLECTION data;
FILTER CASE WHERE .score IN_RANGE ( 0 , 100 )
KEEP OTHERS;

ESEMPIO 12: CASES con Multiple Condizioni
----------------------------------------------
Descrizione: Più branch con CASES
Query:
GET COLLECTION orders;
FILTER CASES
    WHERE .status = 'new' GENERATE ADD FIELDS { .priority : 'high' }
    WHERE .status = 'pending' GENERATE ADD FIELDS { .priority : 'medium' }
    WHERE .status = 'completed' GENERATE ADD FIELDS { .priority : 'low' };

ESEMPIO 13: BUILD con Tutti i Campi Originali
----------------------------------------------
Descrizione: Mantieni tutti campi esistenti
Query:
GET COLLECTION users;
FILTER CASE WHERE .active = true
    GENERATE BUILD { .id, .name, .email, .created_at };

ESEMPIO 14: ADD FIELDS con Stringhe Concatenate
----------------------------------------------
Descrizione: Campo calcolato stringa (valore fisso)
Query:
GET COLLECTION products;
FILTER CASE WHERE .stock > 0
    GENERATE ADD FIELDS { .availability : 'In Stock' };

ESEMPIO 15: REMOVE FIELDS Multipli Privacy
----------------------------------------------
Descrizione: Rimozione campi sensibili multipli
Query:
GET COLLECTION users;
FILTER CASE WHERE WITH .email
    GENERATE REMOVE FIELDS { .password, .ssn, .credit_card, .phone };

ESEMPIO 16: Condizione con Uguaglianza Boolean
----------------------------------------------
Descrizione: Confronto esplicito con true/false
Query:
GET COLLECTION features;
FILTER CASE WHERE .enabled = true AND .beta = false
KEEP OTHERS;

ESEMPIO 17: AND/OR Misti con Precedenza
----------------------------------------------
Descrizione: Precedenza operatori corretta
Query:
GET COLLECTION items;
FILTER CASE 
    WHERE .category = 'A' AND .price > 50 OR .category = 'B' AND .price > 100
KEEP OTHERS;

ESEMPIO 18: IS NOT NULL con AND
----------------------------------------------
Descrizione: Campo non null più condizione
Query:
GET COLLECTION posts;
FILTER CASE 
    WHERE .content IS NOT NULL AND .published = true
KEEP OTHERS;

ESEMPIO 19: Range Sovrapposti (Edge Case)
----------------------------------------------
Descrizione: Due range che si sovrappongono
Query:
GET COLLECTION data;
FILTER CASE 
    WHERE .valueA IN_RANGE [ 0 , 100 ] AND .valueB IN_RANGE [ 50 , 150 ]
KEEP OTHERS;

ESEMPIO 20: WITHOUT Multiple Fields con AND
----------------------------------------------
Descrizione: Assenza di più campi contemporaneamente
Query:
GET COLLECTION incomplete_records;
FILTER CASE 
    WHERE WITHOUT .email AND WITHOUT .phone AND WITHOUT .address
KEEP OTHERS;
