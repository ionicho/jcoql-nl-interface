# ============================================
# JCOQL EXAMPLES - ADVANCED V2 (50 esempi)
# Dataset espanso per RAG + Benchmark LLM
# ============================================

# --- UNION & INTERSECT ---
Titolo: UNION di due collezioni
Descrizione: Unisci clienti premium e clienti VIP
Query: GET COLLECTION premium_customers; UNION GET COLLECTION vip_customers;

Titolo: INTERSECT tra ordini e resi
Descrizione: Trova ordini che hanno anche un reso
Query: GET COLLECTION orders; INTERSECT GET COLLECTION returns;

# --- JOIN Operations ---
Titolo: JOIN tra ordini e prodotti
Descrizione: Unisci informazioni ordini con dettagli prodotti
Query: GET COLLECTION orders; JOIN GET COLLECTION products ON .product_id;

Titolo: LEFT JOIN clienti e ordini
Descrizione: Tutti i clienti, anche quelli senza ordini
Query: GET COLLECTION customers; LEFT JOIN GET COLLECTION orders ON .customer_id;

# --- NESTED Queries ---
Titolo: Filtro con subquery
Descrizione: Prodotti con prezzo maggiore della media
Query: GET COLLECTION products; FILTER CASE WHERE .price > (SELECT AVG(price) FROM products) KEEP OTHERS;

Titolo: IN_RANGE con calcolo dinamico
Descrizione: Età tra 25 e media+10
Query: GET COLLECTION users; FILTER CASE WHERE .age IN_RANGE [25, (SELECT AVG(age)+10 FROM users)] KEEP OTHERS;

# --- AGGREGAZIONI ---
Titolo: GROUP BY categoria
Descrizione: Raggruppa prodotti per categoria e conta
Query: GET COLLECTION products; GROUP BY .category COUNT AS num_products;

Titolo: GROUP con SUM
Descrizione: Totale vendite per venditore
Query: GET COLLECTION sales; GROUP BY .seller_id SUM(.amount) AS total_sales;

Titolo: GROUP con AVG e MIN/MAX
Descrizione: Statistiche prezzi per categoria
Query: GET COLLECTION products; GROUP BY .category AVG(.price) AS avg_price, MIN(.price) AS min_price, MAX(.price) AS max_price;

# --- EXPAND Operations ---
Titolo: EXPAND array tags
Descrizione: Espandi array di tag in documenti separati
Query: GET COLLECTION articles; EXPAND .tags;

Titolo: EXPAND con FILTER
Descrizione: Espandi ordini e filtra per totale > 100
Query: GET COLLECTION customers; EXPAND .orders; FILTER CASE WHERE .total > 100 KEEP OTHERS;

# --- LOOKUP Operations ---
Titolo: LOOKUP dati cliente
Descrizione: Arricchisci ordini con dati cliente
Query: GET COLLECTION orders; LOOKUP .customer_id IN customers AS customer_info;

Titolo: LOOKUP con JOIN
Descrizione: Ordini con dettagli prodotto via lookup
Query: GET COLLECTION orders; LOOKUP .product_id IN products AS product_details; FILTER CASE WHERE .product_details.category = 'Electronics' KEEP OTHERS;

# --- TRAJECTORY (se supportato) ---
Titolo: Trajectory movimenti utente
Descrizione: Traccia percorso utente su mappa
Query: GET COLLECTION user_locations; CREATE TRAJECTORY BY .user_id ORDER BY .timestamp;

Titolo: Trajectory con filtro temporale
Descrizione: Percorsi ultimi 7 giorni
Query: GET COLLECTION gps_data; FILTER CASE WHERE .timestamp > '2024-03-01' KEEP OTHERS; CREATE TRAJECTORY BY .device_id ORDER BY .timestamp;

# --- STRING Operations ---
Titolo: Filtro con CONTAINS
Descrizione: Email contenenti 'gmail'
Query: GET COLLECTION users; FILTER CASE WHERE .email CONTAINS 'gmail' KEEP OTHERS;

Titolo: Filtro STARTS_WITH
Descrizione: Prodotti che iniziano con 'iPhone'
Query: GET COLLECTION products; FILTER CASE WHERE .name STARTS_WITH 'iPhone' KEEP OTHERS;

Titolo: Filtro ENDS_WITH
Descrizione: File che finiscono con .pdf
Query: GET COLLECTION documents; FILTER CASE WHERE .filename ENDS_WITH '.pdf' KEEP OTHERS;

# --- DATE Operations ---
Titolo: Filtro data esatta
Descrizione: Ordini del 15 marzo 2024
Query: GET COLLECTION orders; FILTER CASE WHERE .order_date = '2024-03-15' KEEP OTHERS;

Titolo: Range date
Descrizione: Ordini Q1 2024
Query: GET COLLECTION orders; FILTER CASE WHERE .order_date IN_RANGE ['2024-01-01', '2024-03-31'] KEEP OTHERS;

Titolo: Confronto date
Descrizione: Utenti registrati dopo gennaio 2024
Query: GET COLLECTION users; FILTER CASE WHERE .registration_date > '2024-01-01' KEEP OTHERS;

# --- MULTIPLE CONDITIONS ---
Titolo: AND multipli
Descrizione: Prodotti elettronica, prezzo 100-500, in stock
Query: GET COLLECTION products; FILTER CASE WHERE .category = 'Electronics' AND .price IN_RANGE [100, 500] AND .in_stock = true KEEP OTHERS;

Titolo: OR conditions
Descrizione: Utenti premium o con spesa > 1000
Query: GET COLLECTION users; FILTER CASE WHERE .membership = 'premium' OR .total_spent > 1000 KEEP OTHERS;

Titolo: NOT condition
Descrizione: Prodotti non in categoria 'Toys'
Query: GET COLLECTION products; FILTER CASE WHERE NOT .category = 'Toys' KEEP OTHERS;

# --- NULL Handling ---
Titolo: Filtra NULL esplicito
Descrizione: Documenti con campo note vuoto
Query: GET COLLECTION documents; FILTER CASE WHERE .notes = NULL KEEP OTHERS;

Titolo: Filtra NOT NULL
Descrizione: Utenti con telefono presente
Query: GET COLLECTION users; FILTER CASE WHERE .phone != NULL KEEP OTHERS;

# --- ARRAY Operations ---
Titolo: Array size
Descrizione: Utenti con più di 3 ordini
Query: GET COLLECTION users; FILTER CASE WHERE SIZE(.orders) > 3 KEEP OTHERS;

Titolo: Array contains
Descrizione: Prodotti con tag 'bestseller'
Query: GET COLLECTION products; FILTER CASE WHERE 'bestseller' IN .tags KEEP OTHERS;

# --- NUMERIC Operations ---
Titolo: Modulo matematico
Descrizione: Ordini con ID pari
Query: GET COLLECTION orders; FILTER CASE WHERE .order_id % 2 = 0 KEEP OTHERS;

Titolo: Divisione e confronto
Descrizione: Prodotti con sconto > 20%
Query: GET COLLECTION products; FILTER CASE WHERE (.original_price - .sale_price) / .original_price > 0.2 KEEP OTHERS;

# --- CASE-INSENSITIVE ---
Titolo: Ricerca case-insensitive
Descrizione: Email con 'GMAIL' (ignora maiuscole)
Query: GET COLLECTION users; FILTER CASE WHERE LOWER(.email) CONTAINS 'gmail' KEEP OTHERS;

# --- SORTING ---
Titolo: ORDER BY singolo
Descrizione: Prodotti ordinati per prezzo crescente
Query: GET COLLECTION products; ORDER BY .price ASC;

Titolo: ORDER BY multiplo
Descrizione: Utenti per città e poi per età
Query: GET COLLECTION users; ORDER BY .city ASC, .age DESC;

# --- LIMIT & OFFSET ---
Titolo: TOP 10 prodotti
Descrizione: I 10 prodotti più costosi
Query: GET COLLECTION products; ORDER BY .price DESC; LIMIT 10;

Titolo: Paginazione
Descrizione: Risultati 11-20
Query: GET COLLECTION products; ORDER BY .name ASC; LIMIT 10 OFFSET 10;

# --- PROJECTION ---
Titolo: Seleziona campi specifici
Descrizione: Solo nome ed email utenti
Query: GET COLLECTION users; PROJECT .name, .email;

Titolo: Projection con alias
Descrizione: Rinomina campi nel risultato
Query: GET COLLECTION products; PROJECT .name AS product_name, .price AS cost;

# --- DISTINCT ---
Titolo: Valori unici
Descrizione: Lista città uniche utenti
Query: GET COLLECTION users; DISTINCT .city;

# --- COUNT ---
Titolo: Conta documenti
Descrizione: Numero totale ordini
Query: GET COLLECTION orders; COUNT;

Titolo: Count con filtro
Descrizione: Numero ordini spediti
Query: GET COLLECTION orders; FILTER CASE WHERE .status = 'shipped' KEEP OTHERS; COUNT;

# --- MERGE ---
Titolo: MERGE collezioni
Descrizione: Unisci dati utenti da due sorgenti
Query: GET COLLECTION users_db1; MERGE GET COLLECTION users_db2 ON .email;

# --- SUBTRACT ---
Titolo: SUBTRACT collezioni
Descrizione: Utenti registrati ma senza ordini
Query: GET COLLECTION users; SUBTRACT GET COLLECTION orders ON .user_id;

# --- REGEX (se supportato) ---
Titolo: Pattern matching
Descrizione: Email con dominio valido
Query: GET COLLECTION users; FILTER CASE WHERE .email MATCHES '^[a-z0-9]+@[a-z]+\.[a-z]{2,}$' KEEP OTHERS;

# --- NESTED FIELDS ---
Titolo: Accesso campo nested
Descrizione: Indirizzo città Milano
Query: GET COLLECTION users; FILTER CASE WHERE .address.city = 'Milano' KEEP OTHERS;

Titolo: Array nested
Descrizione: Primo ordine > 100
Query: GET COLLECTION users; FILTER CASE WHERE .orders[0].total > 100 KEEP OTHERS;

# --- BOOLEAN Logic ---
Titolo: XOR condition
Descrizione: Utenti premium O con ordini, ma non entrambi
Query: GET COLLECTION users; FILTER CASE WHERE (.is_premium = true AND SIZE(.orders) = 0) OR (.is_premium = false AND SIZE(.orders) > 0) KEEP OTHERS;

# --- SPECIAL CHARACTERS ---
Titolo: Escape caratteri speciali
Descrizione: Prodotti con apostrofo nel nome
Query: GET COLLECTION products; FILTER CASE WHERE .name CONTAINS 'McDonald\'s' KEEP OTHERS;

# --- GEOSPATIAL (se supportato) ---
Titolo: Filtro coordinate
Descrizione: Luoghi vicino a Milano (lat/lon)
Query: GET COLLECTION places; FILTER CASE WHERE .latitude IN_RANGE [45.4, 45.5] AND .longitude IN_RANGE [9.1, 9.3] KEEP OTHERS;

# --- TIME-BASED ---
Titolo: Filtro ora del giorno
Descrizione: Ordini effettuati dopo le 18:00
Query: GET COLLECTION orders; FILTER CASE WHERE HOUR(.timestamp) >= 18 KEEP OTHERS;

# --- COMPLEX REAL-WORLD ---
Titolo: Dashboard e-commerce
Descrizione: Report vendite prodotti elettronica Q1 2024 per venditore
Query: GET COLLECTION sales; FILTER CASE WHERE .category = 'Electronics' AND .sale_date IN_RANGE ['2024-01-01', '2024-03-31'] KEEP OTHERS; GROUP BY .seller_id SUM(.amount) AS total_revenue, COUNT AS num_sales; ORDER BY total_revenue DESC;

Titolo: Analisi churn clienti
Descrizione: Clienti senza acquisti ultimi 90 giorni
Query: GET COLLECTION customers; FILTER CASE WHERE .last_purchase_date < '2023-12-20' KEEP OTHERS; PROJECT .customer_id, .email, .last_purchase_date;
